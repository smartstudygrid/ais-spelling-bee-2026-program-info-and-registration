📜 Official Rules & Regulations

Authoritative standards governing rounds, scoring, timing, and judge rulings.

1 Point Per Word
1 Scoring System (1 Point)

Each correctly spelt word is awarded exactly 1 point. No penalty points are deducted for misses.

2 Team Consultation Rule

The designated speller may confer with their 2 teammates only during the first 25 seconds of the turn before spelling begins.

3 50-Second Countdown Limit

Total allotment is 50 seconds: 25 seconds for consultation/clarifications (definitions, parts of speech), and 25 seconds to complete active spelling.

4 25-Second Warning Bell

At the 25-second mark, a bell signals that active spelling must commence. Teammates must remain silent; no whispering or coaching is permitted.

5 Word Retracing Policy

A speller may stop and retrace letters already uttered only with the pronouncer's permission, provided the final sequence has not been concluded.

6 UK & US Standards

Both British and American standard orthographies (e.g. colour / color) are fully accepted.

7 Sudden Death Grand Finale

In Round 3, words outside the official booklet can be pronounced until an individual champion is crowned.

8 Finality of Judges

Decisions rendered by the official English Department judging panel are absolute and final.

/**
 * Al Rowad International Schools - Spelling Bee 2026
 * Backend receiver for Student Auditions, Teacher Final Teams & Admin
 */

const TEACHER_SECRET_KEY = "AisSpellingBee2026";
const ADMIN_SECRET_KEY = "Pakistan@1947";

function doPost(e) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const data = JSON.parse(e.postData.contents);
    
    // Admin Verification & Operations
    if (data.type === "ADMIN_ACTION") {
      if (data.passcode !== ADMIN_SECRET_KEY) {
        return createJsonResponse({ status: "error", message: "Unauthorized: Invalid Admin Passcode" });
      }
      return createJsonResponse({ status: "success", message: "Admin authenticated" });
    }

    if (data.type === "STUDENT_AUDITION") {
      let sheet = ss.getSheetByName("Audition_Registrations");
      if (!sheet) {
        sheet = ss.insertSheet("Audition_Registrations");
        sheet.appendRow(["Timestamp", "Class/Grade", "Section", "Full Name", "S-Number", "Division"]);
      }
      
      const division = (data.grade >= 4 && data.grade <= 6) ? "Junior (4-6)" : "Senior (7-9)";
      sheet.appendRow([
        new Date(),
        data.grade,
        data.section,
        data.fullName,
        data.sNumber,
        division
      ]);
      
      return createJsonResponse({ status: "success", message: "Audition registered successfully" });
    }
    
    if (data.type === "TEACHER_FINAL_TEAM") {
      // Validate faculty passcode
      if (data.passcode !== TEACHER_SECRET_KEY) {
        return createJsonResponse({ status: "error", message: "Unauthorized: Invalid Passcode" });
      }
      
      let sheet = ss.getSheetByName("Final_Teams");
      if (!sheet) {
        sheet = ss.insertSheet("Final_Teams");
        sheet.appendRow([
          "Timestamp", "Grade", "Section", "Teacher Name",
          "Student 1 Name", "Student 1 S-Num",
          "Student 2 Name", "Student 2 S-Num",
          "Student 3 Name", "Student 3 S-Num",
          "Division"
        ]);
      }
      
      const division = (data.grade >= 4 && data.grade <= 6) ? "Junior (4-6)" : "Senior (7-9)";
      sheet.appendRow([
        new Date(),
        data.grade,
        data.section,
        data.teacherName,
        data.student1.name, data.student1.sNumber,
        data.student2.name, data.student2.sNumber,
        data.student3.name, data.student3.sNumber,
        division
      ]);
      
      return createJsonResponse({ status: "success", message: "Final team saved successfully" });
    }

    return createJsonResponse({ status: "error", message: "Unknown action type" });

  } catch (err) {
    return createJsonResponse({ status: "error", message: err.toString() });
  }
}

function createJsonResponse(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}
